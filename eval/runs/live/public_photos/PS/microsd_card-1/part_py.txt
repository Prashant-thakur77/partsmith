import cadquery as cq
import math

PARAMS = {
    "thickness": (1.0, 0.5, 2.0, 0.1, "mm", True, "Thickness"),
    "hole_dia": (0.0, 0.0, 3.0, 0.1, "mm", False, "Hidden hole diameter (0=no hole)"),
    "hole_x": (1.3, -7.0, 7.0, 0.1, "mm", False, "Hidden hole X from centre"),
    "hole_y": (0.8, -5.0, 5.0, 0.1, "mm", False, "Hidden hole Y from centre"),
}

# measured base profile polyline points in mm, centred at origin, long axis +X
PROFILE_POINTS = [
    (-7.221, 5.241),
    (-7.629, -4.692),
    (-7.312, -5.535),
    (-6.346, -5.734),
    (-2.557, -5.609),
    (-1.567, -5.007),
    (-0.509, -5.022),
    (-0.129, -5.509),
    (1.006, -5.569),
    (2.018, -5.113),
    (2.698, -4.293),
    (6.823, -4.193),
    (7.502, -3.733),
    (7.627, 4.646),
    (7.192, 5.59),
    (-6.33, 5.685),
]

def build(p: dict) -> cq.Workplane:
    thickness = float(p.get("thickness", 1.0))
    hole_dia = float(p.get("hole_dia", 0.0))
    hole_x = float(p.get("hole_x", 1.3))
    hole_y = float(p.get("hole_y", 0.8))

    # create base profile on XY plane with mounting face at Z=0
    wp = cq.Workplane("XY")
    wp = wp.moveTo(PROFILE_POINTS[0][0], PROFILE_POINTS[0][1])
    for x, y in PROFILE_POINTS[1:]:
        wp = wp.lineTo(x, y)
    wp = wp.close()
    solid = wp.extrude(thickness, combine=True)

    # optional hidden through-hole if indicated by measurement (default 0 = no hole)
    if hole_dia > 0.01:
        r = hole_dia / 2.0
        solid = solid.faces(">Z").workplane().moveTo(hole_x, hole_y).circle(r).cutThruAll()

    return solid

result = build({k: v[0] for k, v in PARAMS.items()})
