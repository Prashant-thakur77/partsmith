def build(p: dict) -> cq.Workplane:
    t = float(p["thickness"])
    hole_d = float(p["hole_dia"])
    hole_cx = float(p["hole_cx"])
    hole_cy = float(p["hole_cy"])
    slot_L = float(p["slot_length"])
    slot_W = float(p["slot_width"])
    slot_angle = float(p["slot_angle"])
    slot_cx = float(p["slot_cx"])
    slot_cy = float(p["slot_cy"])

    # base solid from polyline, extruded from Z=0 to +t (mounting face on Z=0)
    wp = cq.Workplane("XY")
    wp = wp.moveTo(OUTLINE_POINTS[0][0], OUTLINE_POINTS[0][1])
    for x, y in OUTLINE_POINTS[1:]:
        wp = wp.lineTo(x, y)
    wp = wp.close()
    base = wp.extrude(t)

    # central through hole (see-through cylinder)
    base = (
        base.faces(">Z")
        .workplane()
        .pushPoints([(hole_cx, hole_cy)])
        .circle(hole_d / 2.0)
        .cutThruAll()
    )

    # obround slot - length is total length (including end arcs), width is end-arc diameter
    # use transformed workplane + Sketch.slot for reliable stadium shape
    # Sketch.slot(w, h) creates centered stadium with w=total length, h=width
    slot_wp = cq.Workplane("XY").transformed(
        offset=cq.Vector(slot_cx, slot_cy, 0),
        rotate=cq.Vector(0, 0, slot_angle),
    )
    # create slot solid slightly taller than body to ensure through cut
    slot_solid = slot_wp.sketch().slot(slot_L, slot_W).finalize().extrude(t + 0.4)
    # ensure cut is through - slot_solid already spans Z 0..t+0.4
    base = base.cut(slot_solid)

    return base

result = build({k: v[0] for k, v in PARAMS.items()})
