import cadquery as cq
import math

PARAMS = {
    "thickness": (1.35, 0.8, 2.5, 0.05, "mm", True, "Body thickness Z"),
    "xy_offset": (0.15, -0.4, 0.4, 0.05, "mm", True, "XY outline offset for fit"),
    "smooth_iter": (1, 0, 3, 1, "", False, "Smoothing passes"),
}

BASE_PTS = [
    (6.357, -5.843),
    (8.128, -3.373),
    (8.75, -1.139),
    (8.84, 0.318),
    (8.164, 3.351),
    (7.332, 4.826),
    (6.209, 6.169),
    (3.84, 7.759),
    (0.955, 8.474),
    (-2.877, 8.24),
    (-5.091, 7.268),
    (-6.376, 6.282),
    (-8.095, 3.841),
    (-8.604, 2.423),
    (-8.845, 0.491),
    (-8.736, -1.141),
    (-8.354, -2.479),
    (-7.512, -4.17),
    (-6.425, -5.54),
    (-4.698, -6.907),
    (-3.191, -7.693),
    (-1.739, -8.206),
    (0.064, -8.474),
    (2.74, -8.198),
]

def smoothPoints(pts, iters):
    if iters <= 0:
        return pts
    res = pts[:]
    n = len(res)
    for kk in range(iters):
        nxt = []
        for i in range(n):
            x0, y0 = res[i]
            x1, y1 = res[(i - 1) % n]
            x2, y2 = res[(i + 1) % n]
            nx = 0.5 * x0 + 0.25 * x1 + 0.25 * x2
            ny = 0.5 * y0 + 0.25 * y1 + 0.25 * y2
            nxt.append((nx, ny))
        res = nxt
    return res

def polygonCentroid(pts):
    n = len(pts)
    a = 0.0
    cx = 0.0
    cy = 0.0
    for i in range(n):
        x0, y0 = pts[i]
        x1, y1 = pts[(i + 1) % n]
        cross = x0 * y1 - x1 * y0
        a += cross
        cx += (x0 + x1) * cross
        cy += (y0 + y1) * cross
    if abs(a) < 1e-9:
        sx = sum(p[0] for p in pts) / n
        sy = sum(p[1] for p in pts) / n
        return sx, sy
    a *= 0.5
    cx /= (6.0 * a)
    cy /= (6.0 * a)
    return cx, cy

def build(p: dict) -> cq.Workplane:
    thickness = float(p["thickness"])
    xyOffset = float(p.get("xy_offset", 0.0))
    smoothIter = int(p.get("smooth_iter", 1))

    pts = smoothPoints(BASE_PTS, smoothIter)

    cx, cy = polygonCentroid(pts)
    ptsC = [(x - cx, y - cy) for x, y in pts]

    wp = cq.Workplane("XY").spline(ptsC, periodic=True).close()

    if abs(xyOffset) > 1e-9:
        wp = wp.offset2D(xyOffset, kind="arc")

    resultShape = wp.extrude(thickness, combine=True)

    return resultShape

result = build({k: v[0] for k, v in PARAMS.items()})
