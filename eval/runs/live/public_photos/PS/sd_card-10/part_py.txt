import cadquery as cq
import math

MEAS_LENGTH = 31.95
MEAS_WIDTH = 23.678

PARAMS = {
    "length": (31.95, 28.0, 35.0, 0.1, "mm", True, "Overall length X"),
    "width": (23.678, 20.0, 27.0, 0.1, "mm", True, "Overall width Y"),
    "thickness": (2.1, 1.0, 3.5, 0.1, "mm", True, "Body thickness Z"),
    "hidden_hole_1_dia": (0.0, 0.0, 2.5, 0.1, "mm", False, "Hole 1 dia at 3.0,7.5 0=no hole"),
    "hidden_hole_2_dia": (0.0, 0.0, 2.5, 0.1, "mm", False, "Hole 2 dia at 6.7,7.8 0=no hole"),
    "hidden_hole_3_dia": (0.0, 0.0, 2.5, 0.1, "mm", False, "Hole 3 dia at 0.3,0.2 0=no hole"),
    "hidden_hole_4_dia": (0.0, 0.0, 2.5, 0.1, "mm", False, "Hole 4 dia at 5.1,0.2 0=no hole"),
    "rib_count": (0, 0, 5, 1, "", False, "Top rib count topology"),
}

def build(p: dict) -> cq.Workplane:
    length = float(p["length"])
    width = float(p["width"])
    thickness = float(p["thickness"])
    h1 = float(p["hidden_hole_1_dia"])
    h2 = float(p["hidden_hole_2_dia"])
    h3 = float(p["hidden_hole_3_dia"])
    h4 = float(p["hidden_hole_4_dia"])
    ribCount = int(p["rib_count"])

    sx = length / MEAS_LENGTH
    sy = width / MEAS_WIDTH

    basePts = [[-15.459,7.169],[-15.947,-7.312],[-15.886,-9.335],[-15.493,-9.655],[-15.477,-10.075],[-15.975,-10.629],[-15.835,-11.173],[-15.405,-11.593],[-14.681,-11.764],[-11.981,-11.505],[-8.983,-11.773],[-8.035,-11.537],[-6.19,-11.608],[-5.493,-11.306],[-2.263,-11.295],[-0.911,-11.777],[6.878,-11.835],[13.557,-11.437],[14.542,-11.606],[15.292,-11.413],[15.752,-10.974],[15.953,-10.041],[15.783,-8.653],[15.975,-6.647],[15.797,5.267],[15.938,9.627],[15.653,10.341],[15.844,11.199],[15.602,11.737],[15.095,11.37],[14.369,11.843],[13.973,11.525],[12.881,11.71],[12.315,11.289],[12.054,11.645],[10.725,11.625],[10.377,11.418],[9.092,11.819],[8.074,11.557],[5.584,11.434],[3.765,11.649],[2.826,11.16],[2.156,11.19],[1.833,11.622],[-0.305,11.128],[-1.759,11.427],[-2.622,11.307],[-2.898,10.986],[-4.069,11.131],[-4.383,10.738],[-5.311,10.414],[-6.249,10.888],[-6.683,10.851],[-6.937,11.274],[-8.808,11.257],[-9.361,10.506],[-10.024,10.834],[-11.295,10.979],[-11.618,10.612],[-12.211,10.513],[-12.802,9.68],[-14.207,8.544],[-14.323,8.102],[-14.762,7.961]]

    pts = [(x * sx, y * sy) for x, y in basePts]

    wp = cq.Workplane("XY")
    wp = wp.moveTo(pts[0][0], pts[0][1])
    for x, y in pts[1:]:
        wp = wp.lineTo(x, y)
    wp = wp.close()
    result = wp.extrude(thickness, both=False)

    if ribCount > 0:
        ribW = 0.6
        ribH = 0.3
        for i in range(ribCount):
            frac = (i + 1) / (ribCount + 1)
            y = (width * 0.3) * (frac - 0.5)
            ribTop = cq.Workplane("XY").workplane(offset=thickness).moveTo(-length*0.4, y - ribW/2).lineTo(length*0.4, y - ribW/2).lineTo(length*0.4, y + ribW/2).lineTo(-length*0.4, y + ribW/2).close().extrude(ribH, both=False)
            result = result.union(ribTop)

    holeDefs = [
        (h1, 3.0, 7.5),
        (h2, 6.7, 7.8),
        (h3, 0.3, 0.2),
        (h4, 5.1, 0.2),
    ]
    for dia, hx, hy in holeDefs:
        if dia > 0.1:
            hxS = hx * sx
            hyS = hy * sy
            result = result.faces(">Z").workplane().pushPoints([(hxS, hyS)]).circle(dia/2).cutThruAll()

    return result

result = build({k: v[0] for k, v in PARAMS.items()})
