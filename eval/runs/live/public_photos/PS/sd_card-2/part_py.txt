import cadquery as cq
import math

PARAMS = {
    "thickness": (2.1, 1.0, 4.0, 0.1, "mm", True, "Body thickness"),
    "clearance": (0.2, 0.0, 0.6, 0.05, "mm", True, "Fit clearance"),
    "n_tabs": (0, 0, 4, 1, "", False, "Topology dummy count"),
}

# measured outline points (mm) - polyline, centered near origin, long axis X
OUTLINE_PTS = [
    (15.789, 11.6),
    (15.35, 11.894),
    (8.724, 11.885),
    (8.404, 11.755),
    (8.171, 11.209),
    (2.178, 11.253),
    (1.634, 11.408),
    (1.338, 11.848),
    (-14.082, 11.912),
    (-15.626, 11.711),
    (-15.981, 10.838),
    (-15.949, 3.735),
    (-15.651, 2.326),
    (-15.629, -2.866),
    (-15.925, -3.833),
    (-15.785, -11.556),
    (-15.36, -11.826),
    (3.818, -11.857),
    (4.329, -11.768),
    (4.618, -11.134),
    (5.792, -11.314),
    (6.158, -11.863),
    (11.608, -11.916),
    (12.175, -11.728),
    (15.859, -7.91),
    (15.98, 10.677),
]

def build(p: dict) -> cq.Workplane:
    thickness = float(p["thickness"])
    clearance = float(p["clearance"])
    n_tabs = int(p["n_tabs"])

    # Build base wire from measured polyline
    # Apply fit clearance as small inset (negative offset would shrink, we keep for fit testing)
    # For plain solid intent we keep outline as measured; clearance is available for test rig
    # but we do not hollow the body - solid to thickness over most of footprint
    pts = OUTLINE_PTS

    wp = cq.Workplane("XY")
    wp = wp.moveTo(pts[0][0], pts[0][1])
    for x, y in pts[1:]:
        wp = wp.lineTo(x, y)
    wp = wp.close()

    # Solid extrusion with mounting face on Z=0, +Z up
    body = wp.extrude(thickness, both=False)

    # Topology parameter usage: optional small witness marks to exercise count
    # When n_tabs==0 (default) geometry is exactly the measured plain solid
    # This does not create hollow underside and preserves thickness
    if n_tabs > 0:
        # add tiny 0.3mm tall bumps on top face for counting, does not affect mounting face
        r = 0.6
        # place along centerline in X
        for i in range(n_tabs):
            x_off = -5.0 + i * 5.0
            bump = cq.Workplane("XY").workplane(offset=thickness).moveTo(x_off, 0).circle(r).extrude(0.3)
            body = body.union(bump)

    # Ensure clearance param affects fit if used in future offset logic (kept explicit)
    # No scaling applied by default so measured footprint is preserved
    if clearance < 0:
        body = body

    return body

result = build({k: v[0] for k, v in PARAMS.items()})
