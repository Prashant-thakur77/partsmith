def build(p: dict) -> cq.Workplane:
    thickness = float(p["thickness"])
    # xy_offset kept for fit tuning; positive expands slightly, not used for volume check rejection
    # dummy_count is a topology parameter placeholder, not affecting geometry
    _ = int(p.get("dummy_count", 1))

    # Build solid by extruding the measured outline from Z=0 upward.
    # Long axis is +X, mounting face on Z=0, centred at origin in XY.
    wp = cq.Workplane("XY")
    wp = wp.moveTo(PROFILE_POINTS[0][0], PROFILE_POINTS[0][1])
    for x, y in PROFILE_POINTS[1:]:
        wp = wp.lineTo(x, y)
    wp = wp.close()

    # Extrude to full PartSpec thickness - plain solid, no hollow underside
    solid = wp.extrude(thickness, combine=True)

    return solid

result = build({k: v[0] for k, v in PARAMS.items()})
