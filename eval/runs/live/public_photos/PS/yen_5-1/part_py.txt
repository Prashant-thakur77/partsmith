import cadquery as cq
import math

PARAMS = {
    "thickness": (1.5, 0.8, 3.0, 0.1, "mm", False, "Thickness"),
    "hole_diameter": (21.628, 18.0, 25.0, 0.1, "mm", True, "Hole Diameter"),
    "outline_scale": (1.15, 0.85, 1.15, 0.01, "", False, "Outline Scale"),
    "hole_count": (1, 1, 1, 1, "", False, "Hole Count"),
}

]

# measured hole center (essentially origin)
HOLE_CENTER = (0.033, -0.004)

def build(p: dict) -> cq.Workplane:
    scale = p["outline_scale"]
    thick = p["thickness"]
    hd = p["hole_diameter"]
    hcount = int(p["hole_count"])

    scaled_pts = [(x * scale, y * scale) for x, y in BASE_POINTS]
    scaled_center = (HOLE_CENTER[0] * scale, HOLE_CENTER[1] * scale)

    # solid body extruded from mounting face Z=0 upward, centred at origin XY
    body = cq.Workplane("XY").polyline(scaled_pts).close().extrude(thick)

    # through hole - plain solid, cut through entire thickness
    if hcount >= 1 and hd > 0:
        body = (
            body.faces(">Z")
            .workplane()
            .center(scaled_center[0], scaled_center[1])
            .circle(hd / 2.0)
            .cutThruAll()
        )

    return body

result = build({k: v[0] for k, v in PARAMS.items()})
