import cadquery as cq
import math

PARAMS = {
    "thickness": (1.0, 0.6, 2.0, 0.1, "mm", True, "Body thickness"),
    "scale_x": (1.0, 0.9, 1.1, 0.01, "mm", True, "Outline scale X"),
    "scale_y": (1.0, 0.9, 1.1, 0.01, "mm", True, "Outline scale Y"),
    "detail_level": (0.0, 0, 2, 1, "mm", False, "Detail level count"),
}

baseProfilePoints = [
    (-7.254, -4.504),
    (-6.726, -5.021),
    (-5.875, -5.222),
    (-5.212, -4.684),
    (-4.924, -5.223),
    (-1.76, -5.23),
    (-0.767, -4.621),
    (-0.159, -5.035),
    (1.377, -5.185),
    (3.008, -4.013),
    (7.108, -3.822),
    (7.452, -3.088),
    (7.339, 4.845),
    (6.454, 5.229),
    (-5.688, 5.255),
    (-7.036, 5.005),
    (-7.418, 4.453),
]

def build(p: dict) -> cq.Workplane:
    thickness = float(p.get("thickness", 1.0))
    scaleX = float(p.get("scale_x", 1.0))
    scaleY = float(p.get("scale_y", 1.0))
    detailLevel = int(p.get("detail_level", 0))

    pts = [(x * scaleX, y * scaleY) for x, y in baseProfilePoints]

    xs = [x for x, y in pts]
    ys = [y for x, y in pts]
    cx = (min(xs) + max(xs)) * 0.5
    cy = (min(ys) + max(ys)) * 0.5
    pts = [(x - cx, y - cy) for x, y in pts]

    wp = cq.Workplane("XY")
    wp = wp.moveTo(pts[0][0], pts[0][1])
    for x, y in pts[1:]:
        wp = wp.lineTo(x, y)
    wp = wp.close()
    solid = wp.extrude(thickness, combine=True)

    if detailLevel > 0:
        try:
            cham = min(0.2 * detailLevel, thickness * 0.4)
            solid = solid.edges(">Z").chamfer(cham)
        except Exception:
            pass

    return solid

result = build({k: v[0] for k, v in PARAMS.items()})
