import cadquery as cq
import math

PARAMS = {
    "brick_length": (31.7, 30.0, 33.0, 0.1, "mm", False, "Brick length X"),
    "brick_width": (15.9, 14.0, 17.0, 0.1, "mm", False, "Brick width Y"),
    "brick_height": (9.9, 8.0, 11.0, 0.1, "mm", False, "Body thickness Z"),
    "stud_diameter": (4.8, 4.0, 5.5, 0.1, "mm", True, "Stud outer diameter"),
    "stud_height": (1.5, 1.2, 2.5, 0.1, "mm", False, "Stud height"),
    "stud_pitch": (8.0, 7.0, 9.0, 0.1, "mm", False, "Stud pitch 8mm"),
    "studs_x": (4, 1, 6, 1, "", False, "Studs along X (long axis)"),
    "studs_y": (2, 1, 4, 1, "", False, "Studs along Y (short axis)"),
    "edge_fillet": (0.3, 0.0, 1.0, 0.1, "mm", False, "Top edge fillet radius"),
}

def build(p: dict) -> cq.Workplane:
    length = float(p["brick_length"])
    width = float(p["brick_width"])
    height = float(p["brick_height"])
    sd = float(p["stud_diameter"])
    sh = float(p["stud_height"])
    pitch = float(p["stud_pitch"])
    nx = int(p["studs_x"])
    ny = int(p["studs_y"])

    # Solid body, mounting face on Z=0, centered at origin, long axis +X
    body = cq.Workplane("XY").rect(length, width).extrude(height)

    # Studs on top face - 2x4 array on 8mm pitch, centered
    studs = None
    for ix in range(nx):
        x = (ix - (nx - 1) / 2.0) * pitch
        for iy in range(ny):
            y = (iy - (ny - 1) / 2.0) * pitch
            s = cq.Workplane("XY").workplane(offset=height).moveTo(x, y).circle(sd / 2.0).extrude(sh)
            if studs is None:
                studs = s
            else:
                studs = studs.union(s)

    result = body.union(studs) if studs is not None else body

    ef = float(p["edge_fillet"])
    if ef > 0.001:
        try:
            result = result.edges("|Z and >Z").fillet(ef)
        except Exception:
            pass

    return result

result = build({k: v[0] for k, v in PARAMS.items()})
