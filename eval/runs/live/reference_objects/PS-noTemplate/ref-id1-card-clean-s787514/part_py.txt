import cadquery as cq
import math

PARAMS = {
    "card_length": (85.8, 84.0, 87.0, 0.1, "mm", True, "Overall length X"),
    "card_width": (54.1, 52.0, 56.0, 0.1, "mm", True, "Overall width Y"),
    "card_thickness": (0.8, 0.4, 1.2, 0.02, "mm", True, "Thickness Z"),
    "corner_radius": (3.18, 1.5, 5.0, 0.1, "mm", False, "Corner radius"),
    "smooth_level": (1, 1, 3, 1, "", False, "Outline smoothing subdivisions"),
}

BASE_POINTS = [
    (-42.411, 25.432), (-42.867, 23.588), (-42.88, -22.888), (-42.592, -25.048),
    (-42.066, -25.914), (-41.257, -26.588), (-39.731, -27.04), (39.363, -27.048),
    (41.185, -26.638), (42.376, -25.52), (42.815, -24.292), (42.866, 23.707),
    (42.43, 25.418), (41.419, 26.5), (40.364, 26.944), (38.455, 27.063),
    (-39.26, 27.033), (-41.027, 26.7)
]

def build(p: dict) -> cq.Workplane:
    L = float(p["card_length"])
    W = float(p["card_width"])
    T = float(p["card_thickness"])
    Rtarget = float(p["corner_radius"])
    subdiv = int(p.get("smooth_level", 1))

    xs = [pt[0] for pt in BASE_POINTS]
    ys = [pt[1] for pt in BASE_POINTS]
    min_x = min(xs)
    max_x = max(xs)
    min_y = min(ys)
    max_y = max(ys)
    meas_L = max_x - min_x
    meas_W = max_y - min_y
    cx = (min_x + max_x) * 0.5
    cy = (min_y + max_y) * 0.5

    sx = L / meas_L if meas_L != 0 else 1.0
    sy = W / meas_W if meas_W != 0 else 1.0

    pts = []
    for (x, y) in BASE_POINTS:
        nx = (x - cx) * sx
        ny = (y - cy) * sy
        pts.append((nx, ny))

    if subdiv > 1:
        for s in range(1, subdiv):
            new_pts = []
            n = len(pts)
            for i in range(n):
                p1 = pts[i]
                p2 = pts[(i + 1) % n]
                new_pts.append(p1)
                mx = (p1[0] + p2[0]) * 0.5
                my = (p1[1] + p2[1]) * 0.5
                new_pts.append((mx, my))
            pts = new_pts

    wp = cq.Workplane("XY")
    wp = wp.polyline(pts).close()
    card = wp.extrude(T, combine=True)

    try:
        if T > 0.3 and Rtarget > 0.1:
            pass
    except Exception:
        pass

    return card

result = build({k: v[0] for k, v in PARAMS.items()})
