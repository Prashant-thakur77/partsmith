import cadquery as cq
import math

PARAMS = {
    "card_length": (85.669, 80.0, 90.0, 0.1, "mm", True, "Card length X (long axis)"),
    "card_width": (54.042, 50.0, 60.0, 0.1, "mm", True, "Card width Y (short axis)"),
    "thickness": (0.8, 0.4, 1.5, 0.05, "mm", False, "Card thickness Z"),
    "corner_radius": (3.18, 1.0, 5.0, 0.1, "mm", False, "Corner fillet radius"),
    "outline_points": (16, 8, 32, 1, "", False, "Outline polyline resolution"),
}

def build(p: dict) -> cq.Workplane:
    W = float(p["card_length"])
    H = float(p["card_width"])
    T = float(p["thickness"])
    R = float(p["corner_radius"])

    maxR = min(W, H) * 0.5 - 0.05
    if R > maxR:
        R = maxR
    if R < 0.1:
        R = 0.1

    # Base solid: centred at origin in XY, mounting face on Z=0, +Z up, long axis +X
    base = cq.Workplane("XY").rect(W, H).extrude(T, combine=True)

    # Rounded corners in XY plane via vertical edge fillet - reproduces measured polyline
    try:
        base = base.edges("|Z").fillet(R)
    except Exception:
        w2 = W * 0.5
        h2 = H * 0.5
        r = R
        k = 1 - 0.70710678
        base = (
            cq.Workplane("XY")
            .moveTo(-w2 + r, -h2)
            .lineTo(w2 - r, -h2)
            .threePointArc((w2, -h2 + r), (w2 - r * k, -h2 + r * k))
            .lineTo(w2, h2 - r)
            .threePointArc((w2 - r, h2), (w2 - r * k, h2 - r * k))
            .lineTo(-w2 + r, h2)
            .threePointArc((-w2, h2 - r), (-w2 + r * k, h2 - r * k))
            .lineTo(-w2, -h2 + r)
            .threePointArc((-w2 + r, -h2), (-w2 + r * k, -h2 + r * k))
            .close()
            .extrude(T)
        )

    return base

result = build({k: v[0] for k, v in PARAMS.items()})
