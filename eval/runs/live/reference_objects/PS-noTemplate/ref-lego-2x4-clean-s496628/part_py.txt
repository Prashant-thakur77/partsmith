import cadquery as cq
import math

PARAMS = {
    "brick_length": (31.8, 28.0, 36.0, 0.1, "mm", True, "Length X"),
    "brick_width": (15.8, 12.0, 20.0, 0.1, "mm", True, "Width Y"),
    "brick_height": (9.9, 8.0, 12.0, 0.1, "mm", True, "Body height Z"),
    "stud_dia": (4.9, 4.0, 5.2, 0.1, "mm", True, "Stud outer diameter"),
    "stud_height": (1.3, 1.2, 2.4, 0.1, "mm", True, "Stud height"),
    "stud_pitch": (8.0, 7.0, 9.0, 0.1, "mm", True, "Stud pitch"),
    "nx": (4, 1, 6, 1, "", False, "Studs along X"),
    "ny": (2, 1, 4, 1, "", False, "Studs along Y"),
}

def build(p: dict) -> cq.Workplane:
    L = float(p["brick_length"])
    W = float(p["brick_width"])
    H = float(p["brick_height"])
    sd = float(p["stud_dia"])
    sh = float(p["stud_height"])
    pitch = float(p["stud_pitch"])
    nx = int(p["nx"])
    ny = int(p["ny"])

    # Body solid to thickness, centered at origin, mounting face on Z=0, long axis +X
    body = cq.Workplane("XY").rect(L, W).extrude(H)

    # Studs on top face
    # centered grid: spacing = pitch
    if nx > 1:
        x0 = - (nx - 1) * pitch / 2.0
    else:
        x0 = 0.0
    if ny > 1:
        y0 = - (ny - 1) * pitch / 2.0
    else:
        y0 = 0.0

    studs = None
    for ix in range(nx):
        x = x0 + ix * pitch
        for iy in range(ny):
            y = y0 + iy * pitch
            s = cq.Workplane("XY").workplane(offset=H).moveTo(x, y).circle(sd / 2.0).extrude(sh)
            if studs is None:
                studs = s
            else:
                studs = studs.union(s)

    if studs is not None:
        result = body.union(studs)
    else:
        result = body

    return result

result = build({k: v[0] for k, v in PARAMS.items()})
