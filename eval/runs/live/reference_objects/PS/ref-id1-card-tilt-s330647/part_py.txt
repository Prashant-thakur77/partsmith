import cadquery as cq
import math

PARAMS = {
    "card_length": (85.6, 80.0, 90.0, 0.1, "mm", True, "Length"),
    "card_width": (53.98, 50.0, 58.0, 0.1, "mm", True, "Width"),
    "card_thickness": (0.8, 0.5, 1.2, 0.01, "mm", True, "Thickness"),
    "corner_radius": (3.18, 1.5, 5.0, 0.1, "mm", False, "Corner radius"),
    "resolution": (32, 8, 64, 1, "", False, "Facet resolution"),
}

def build(p: dict) -> cq.Workplane:
    L = float(p["card_length"])
    W = float(p["card_width"])
    T = float(p["card_thickness"])
    R = float(p["corner_radius"])
    # clamp radius to be valid for fillet (must be < half of smallest side)
    max_r = min(L, W) * 0.49
    if R > max_r:
        R = max_r
    if R < 0.1:
        R = 0.1

    # Base card: centred at origin in XY, mounting face on Z=0, long axis +X, +Z up
    # Use 2D rectangle on XY plane and extrude from Z=0 to Z=T
    base = (
        cq.Workplane("XY")
        .rect(L, W, centered=True)
        .extrude(T, combine=True)
    )

    # Rounded corners - fillet vertical edges
    # This reproduces the measured polyline rounded corners
    try:
        base = base.edges("|Z").fillet(R)
    except Exception:
        # fallback if fillet fails due to geometry tolerance
        pass

    return base

result = build({k: v[0] for k, v in PARAMS.items()})
