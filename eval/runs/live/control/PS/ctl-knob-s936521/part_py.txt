PROFILE_POINTS = [
    (3.505, -13.584), (4.36, -12.637), (4.991, -12.489), (5.965, -12.687), (8.019, -11.455),
    (8.498, -10.271), (10.038, -9.775), (11.428, -8.125), (11.572, -6.644), (12.833, -5.665),
    (13.598, -3.549), (13.202, -2.192), (13.443, -1.541), (14.032, -0.943), (14.018, 1.516),
    (13.18, 2.898), (13.547, 4.159), (12.634, 6.471), (11.273, 8.628), (10.092, 9.941),
    (9.049, 10.27), (8.188, 11.523), (6.007, 12.779), (4.493, 12.832), (3.621, 13.631),
    (1.169, 14.035), (-0.172, 13.626), (-1.422, 14.028), (-3.371, 13.687), (-4.709, 12.827),
    (-6.048, 12.711), (-7.886, 11.673), (-8.887, 10.396), (-10.118, 9.794), (-11.518, 8.1),
    (-11.814, 6.925), (-12.79, 5.868), (-13.615, 3.563), (-13.473, 2.335), (-14.046, 1.111),
    (-14.047, -1.259), (-13.433, -2.379), (-13.589, -3.792), (-12.771, -6.045), (-11.727, -6.922),

]

PARAMS = {
    "thickness": (16.0, 10.0, 22.0, 0.5, "mm", False, "Body thickness / height"),
    "bore_dia": (6.0, 4.0, 8.0, 0.1, "mm", True, "D-shaft bore diameter"),
    "flat_depth": (1.0, 0.3, 2.0, 0.1, "mm", True, "Flat cut depth from OD"),
    "bore_depth": (13.0, 6.0, 16.0, 0.5, "mm", True, "Blind bore depth from bottom face"),
    "knurl_count": (36, 16, 72, 1, "", False, "Knurl count topology reference"),
}

def build(p: dict) -> cq.Workplane:
    thickness = float(p["thickness"])
    bore_dia = float(p["bore_dia"])
    flat_depth = float(p["flat_depth"])
    bore_depth = float(p["bore_depth"])
    knurl_count = int(p.get("knurl_count", 36))

    if bore_depth > thickness - 0.1:
        bore_depth = thickness - 0.1
    if bore_depth < 0.5:
        bore_depth = 0.5

    r = bore_dia * 0.5
    fy = r - flat_depth
    if fy > r:
        fy = r - 0.01
    if fy < -r:
        fy = -r + 0.01

    x0_sq = r * r - fy * fy
    if x0_sq < 0:
        x0_sq = 0
    x0 = math.sqrt(x0_sq)

    wp = cq.Workplane("XY")
    body = wp.polyline(PROFILE_POINTS).close().extrude(thickness, combine=True)

    try:
        body = body.faces(">Z").chamfer(0.4)
    except Exception:
        pass

    if x0 < 0.01 and knurl_count > 0:
        cutter = cq.Workplane("XY").circle(r).extrude(bore_depth)
    else:
        cutter = (
            cq.Workplane("XY")
            .moveTo(x0, fy)
            .lineTo(-x0, fy)
            .threePointArc((0, -r), (x0, fy))
            .close()
            .extrude(bore_depth)
        )

    body = body.cut(cutter)
    return body

result = build({k: v[0] for k, v in PARAMS.items()})
