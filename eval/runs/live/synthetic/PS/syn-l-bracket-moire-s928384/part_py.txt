import cadquery as cq
import math

PARAMS = {
    "thickness": (3.5, 1.0, 6.0, 0.1, "mm", True, "Base plate thickness"),
    "flange_height": (18.0, 10.0, 35.0, 0.5, "mm", False, "Flange height from mounting face"),
    "flange_thickness": (3.5, 1.5, 6.0, 0.1, "mm", True, "Flange thickness (X)"),
    "flange_width": (23.46, 18.0, 28.0, 0.1, "mm", False, "Flange width in Y"),
    "flange_hole_dia": (5.0, 3.0, 8.0, 0.1, "mm", True, "Flange mounting hole diameter"),
    "flange_hole_z": (9.0, 5.0, 20.0, 0.1, "mm", False, "Flange hole Z height from base face"),
    "flange_hole_count": (1, 1, 3, 1, "", False, "Number of flange holes"),
    "base_hole_dia": (5.643, 4.0, 7.0, 0.01, "mm", True, "Base through hole diameter"),
    "base_hole_x": (2.744, -5.0, 8.0, 0.01, "mm", False, "Base hole X"),
    "base_hole_y": (0.134, -5.0, 5.0, 0.01, "mm", False, "Base hole Y"),
    "bend_fillet_r": (1.5, 0.0, 3.0, 0.1, "mm", False, "Inside bend fillet radius"),
}

def build(p: dict) -> cq.Workplane:
    thickness = float(p["thickness"])
    flange_height = float(p["flange_height"])
    flange_thickness = float(p["flange_thickness"])
    flange_width = float(p["flange_width"])
    flange_hole_dia = float(p["flange_hole_dia"])
    flange_hole_z = float(p["flange_hole_z"])
    flange_hole_count = int(p["flange_hole_count"])
    base_hole_dia = float(p["base_hole_dia"])
    base_hole_x = float(p["base_hole_x"])
    base_hole_y = float(p["base_hole_y"])

    pts = [
        (-17.758, 11.839),
        (-17.732, -11.626),
        (-16.803, -12.488),
        (-12.879, -12.506),
        (-12.728, -12.064),
        (-3.215, -11.573),
        (15.648, -11.582),
        (16.634, -11.282),
        (17.383, -10.618),
        (17.737, -9.875),
        (17.825, 9.713),
        (17.382, 10.889),
        (16.627, 11.565),
        (15.564, 11.861),
        (-3.292, 11.843),
        (-12.663, 12.162),
        (-12.943, 12.52),
        (-16.691, 12.538),
    ]

    # base plate - solid to thickness over most of footprint
    base = cq.Workplane("XY").polyline(pts).close().extrude(thickness, combine=True)

    # L-flange at the flat -X edge (mount perpendicular to base)
    # Mounting face on Z=0, long axis +X, centred at origin, +Z up
    x0 = -17.758  # min X edge from base_profile
    # flange solid from Z=0 up to flange_height
    flange = (
        cq.Workplane("XY")
        .workplane(offset=0)
        .moveTo(x0 + flange_thickness / 2.0, 0)
        .rect(flange_thickness, flange_width, centered=True)
        .extrude(flange_height, combine=False)
    )

    result = base.union(flange)

    # inside bend fillet along the bend line (X direction)
    # add fillet to the interior edge if radius > 0
    # attempt edge fillet, ignore failure on geometry limits
    bend_r = float(p["bend_fillet_r"])
    if bend_r > 0.01:
        try:
            # select vertical interior edge at junction of base and flange
            # edge running in Y at X ~ x0+flange_thickness, Z ~ thickness
            result = result.edges(cq.selectors.NearestToPointSelector((x0 + flange_thickness, 0, thickness))).fillet(bend_r)
        except Exception:
            pass

    # base through hole (vertical cylinder)
    result = (
        result.faces(">Z")
        .workplane(origin=(0, 0, 0))
        .moveTo(base_hole_x, base_hole_y)
        .circle(base_hole_dia / 2.0)
        .cutThruAll()
    )

    # flange hole(s) - through thickness in X direction
    # placed on the -X face workplane, centered in Y, at height flange_hole_z
    if flange_hole_count >= 1:
        # single centered or distributed in Y if count >1
        if flange_hole_count == 1:
            ys = [0.0]
        else:
            # spread across flange_width with margin
            span = flange_width * 0.6
            step = span / (flange_hole_count - 1) if flange_hole_count > 1 else 0
            start = -span / 2.0
            ys = [start + i * step for i in range(flange_hole_count)]
        for y_c in ys:
            # use YZ workplane offset at flange outer face and cut in X
            # create tool via face selector <X
            try:
                result = (
                    result.faces("<X")
                    .workplane(origin=(0, 0, 0))
                    .moveTo(y_c, flange_hole_z)
                    .circle(flange_hole_dia / 2.0)
                    .cutThruAll()
                )
            except Exception:
                # fallback: create cylinder tool
                tool = (
                    cq.Workplane("YZ")
                    .workplane(offset=x0 - 2.0)
                    .moveTo(y_c, flange_hole_z)
                    .circle(flange_hole_dia / 2.0)
                    .extrude(flange_thickness + 4.0)
                )
                result = result.cut(tool)

    return result

result = build({k: v[0] for k, v in PARAMS.items()})
