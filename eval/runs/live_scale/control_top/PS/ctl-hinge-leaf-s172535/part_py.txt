def build(p: dict) -> cq.Workplane:
    thickness = float(p.get("thickness", 2.0))
    clearance = float(p.get("clearance_mm", 0.2))
    n = int(round(float(p.get("num_holes", 4))))

    # Solid leaf body extruded from Z=0 up to thickness, long axis +X
    # Base profile is closed polyline in XY plane
    base = cq.Workplane("XY").polyline(BASE_POINTS).close().extrude(thickness)

    # Collect hole definitions
    holes = [
        (float(p["hole_1_x"]), float(p["hole_1_y"]), float(p["hole_1_dia"])),
        (float(p["hole_2_x"]), float(p["hole_2_y"]), float(p["hole_2_dia"])),
        (float(p["hole_3_x"]), float(p["hole_3_y"]), float(p["hole_3_dia"])),
        (float(p["hole_4_x"]), float(p["hole_4_y"]), float(p["hole_4_dia"])),
    ]

    # Through holes along -Z, clearance added to diameter for fit
    # Workplane on top face at Z=thickness cutting downward
    for i in range(min(n, len(holes))):
        hx, hy, hd = holes[i]
        d = hd + clearance
        # cutThruAll works from top face; ensure cutter goes through entire thickness
        base = base.faces(">Z").workplane().pushPoints([(hx, hy)]).hole(d)

    return base

result = build({k: v[0] for k, v in PARAMS.items()})
