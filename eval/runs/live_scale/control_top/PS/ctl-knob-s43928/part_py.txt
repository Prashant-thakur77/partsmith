PARAMS = {
    "thickness": (14.0, 8.0, 20.0, 0.5, "mm", False, "Body thickness"),
    "bore_dia": (6.0, 3.0, 10.0, 0.1, "mm", True, "Mating D-shaft diameter"),
    "bore_flat_depth": (1.0, 0.2, 2.5, 0.1, "mm", True, "Flat depth (chord height removed)"),
    "bore_depth": (10.0, 5.0, 14.0, 0.5, "mm", True, "Bore depth from mounting face (14=through)"),
    "clearance": (0.2, 0.0, 0.6, 0.05, "mm", True, "Diametric clearance for PLA FDM"),
    "slot_length": (6.0, 3.0, 10.0, 0.1, "mm", True, "D-slot length (inferred)"),
    "slot_width": (1.0, 0.4, 2.0, 0.1, "mm", True, "D-slot width (inferred)"),
    "fillet_r": (1.0, 0.2, 2.0, 0.2, "mm", False, "Edge fillet radius"),
    "knurl_smooth": (1, 0, 3, 1, "", False, "Topology smoothing iterations (integer)"),
}

def build(p: dict) -> cq.Workplane:
    thickness = float(p["thickness"])
    bore_dia = float(p["bore_dia"])
    flat_depth = float(p["bore_flat_depth"])
    bore_depth = float(p["bore_depth"])
    clearance = float(p["clearance"])
    fillet_r = float(p["fillet_r"])

    # Effective bore radius with clearance
    r = (bore_dia + clearance) / 2.0
    # clamp flat depth
    if flat_depth >= 2*r - 0.1:
        flat_depth = 2*r - 0.2
    if flat_depth < 0.05:
        flat_depth = 0.05
    flat_y = r - flat_depth  # flat line y = flat_y, cap y>flat_y removed
    # intersection half width
    # if flat_y exceeds r, no cut
    if abs(flat_y) > r:
        x_int = 0.0
    else:
        x_int = math.sqrt(max(0.0, r*r - flat_y*flat_y))

    # Base solid: spline footprint extruded from Z=0 upward +Z, centered in XY
    # Long axis +X is natural as profile is roughly symmetric
    pts = OUTLINE_PTS
    # close spline explicitly
    body = cq.Workplane("XY").spline(pts, periodic=True).close().extrude(thickness, combine=True)

    # Fillet top perimeter slightly for feel - fillet_all intent
    # Use XY top edges
    try:
        if fillet_r > 0.05:
            # fillet top and bottom outer edges; use edges with Z normals
            body = body.edges(cq.selectors.BoxSelector(( -20,-20, thickness-0.1),(20,20,thickness+0.1))).fillet(min(fillet_r, 1.2))
    except Exception:
        pass
    try:
        if fillet_r > 0.05:
            body = body.edges(cq.selectors.BoxSelector((-20,-20,-0.1),(20,20,0.1))).fillet(min(fillet_r, 0.8))
    except Exception:
        pass

    # D-shaft bore: D profile centered at origin
    # Build D wire: start at -x_int,flat_y -> +x_int,flat_y -> arc around bottom to close
    # Angle for arc endpoints
    if r > 1e-6 and abs(flat_y) <= r:
        # use D shape
        d_bore = (
            cq.Workplane("XY")
            .moveTo(-x_int, flat_y)
            .lineTo(x_int, flat_y)
            # threePointArc needs mid point at bottom
            .threePointArc((0, -r), (-x_int, flat_y))
            .close()
            .extrude(bore_depth, combine=False)
        )
    else:
        # fallback circular
        d_bore = cq.Workplane("XY").circle(r).extrude(bore_depth, combine=False)

    # Ensure bore sits from mounting face Z=0 upward: extrusion already from 0 to bore_depth
    # If bore_depth >= thickness treat as through, ensure cut through
    if bore_depth >= thickness - 1e-6:
        # make slightly longer to guarantee through cut
        if r > 1e-6 and abs(flat_y) <= r:
            d_bore = (
                cq.Workplane("XY")
                .moveTo(-x_int, flat_y)
                .lineTo(x_int, flat_y)
                .threePointArc((0, -r), (-x_int, flat_y))
                .close()
                .extrude(thickness + 0.4, combine=False)
            )
        else:
            d_bore = cq.Workplane("XY").circle(r).extrude(thickness + 0.4, combine=False)

    # Cut bore from body
    body = body.cut(d_bore)

    # Bore mouth chamfer/fillet for printability (radius fillet_all id d_shaft_bore_1)
    try:
        if fillet_r > 0.05:
            # select bore edges at Z=0
            body = body.edges(cq.selectors.NearestToPointSelector((0,0,0))).fillet(min(fillet_r*0.6, 0.6))
    except Exception:
        pass

    # Center at origin in XY already; ensure mounting face on Z=0 (already)
    return body

result = build({k: v[0] for k, v in PARAMS.items()})
