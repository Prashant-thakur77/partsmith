import cadquery as cq
import math

PARAMS = {
    "brick_length": (31.8, 30.0, 33.0, 0.1, "mm", True, "Brick length X"),
    "brick_width": (15.8, 14.0, 17.0, 0.1, "mm", True, "Brick width Y"),
    "brick_height": (9.7, 8.0, 11.0, 0.1, "mm", False, "Brick body height"),
    "stud_dia": (4.6, 4.0, 5.5, 0.1, "mm", True, "Stud outer diameter"),
    "stud_height": (1.7, 1.0, 2.5, 0.1, "mm", False, "Stud height"),
    "stud_pitch": (8.0, 7.0, 9.0, 0.1, "mm", True, "Stud pitch"),
    "n_studs_x": (4, 1, 6, 1, "", False, "Studs along X"),
    "n_studs_y": (2, 1, 4, 1, "", False, "Studs along Y"),
}

def build(p: dict) -> cq.Workplane:
    L = float(p["brick_length"])
    W = float(p["brick_width"])
    H = float(p["brick_height"])
    d = float(p["stud_dia"])
    hs = float(p["stud_height"])
    pitch = float(p["stud_pitch"])
    nx = int(p["n_studs_x"])
    ny = int(p["n_studs_y"])

    # Body solid to thickness, centred in X/Y, bottom on Z=0
    body = cq.Workplane("XY").rect(L, W, centered=True).extrude(H)

    # Centres for 2x4 array on 8mm pitch, centred at origin
    # For nx=4, ny=2 this gives x=-12,-4,4,12 y=-4,4 matching spec
    if nx > 1:
        x_start = - (nx - 1) * pitch / 2.0
    else:
        x_start = 0.0
    if ny > 1:
        y_start = - (ny - 1) * pitch / 2.0
    else:
        y_start = 0.0

    r = d / 2.0
    result = body
    for ix in range(nx):
        x = x_start + ix * pitch
        for iy in range(ny):
            y = y_start + iy * pitch
            stud = cq.Workplane("XY").workplane(offset=H).moveTo(x, y).circle(r).extrude(hs)
            result = result.union(stud)

    return result

result = build({k: v[0] for k, v in PARAMS.items()})
