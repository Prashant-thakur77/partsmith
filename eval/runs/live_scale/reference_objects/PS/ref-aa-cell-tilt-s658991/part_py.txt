import cadquery as cq
import math

PARAMS = {
    "thickness": (14.5, 10.0, 20.0, 0.1, "mm", True, "Body thickness (+Z)"),
    "length_scale": (1.0, 0.9, 1.1, 0.01, "", False, "Length scale X"),
    "width_scale": (1.0, 0.9, 1.1, 0.01, "", False, "Width scale Y"),
    "smooth_count": (1, 1, 3, 1, "", False, "Outline smoothing passes"),
}

# measured polyline footprint, AA cell lying on side, long axis +X, centered at origin
BASE_POINTS = [
    [24.727,-2.219],
    [24.687,2.649],
    [24.371,6.147],
    [24.115,6.77],
    [23.678,7.052],
    [-23.187,7.1],
    [-23.689,6.531],
    [-24.152,5.346],
    [-24.733,1.111],
    [-24.505,-2.915],
    [-23.796,-6.218],
    [-22.985,-6.985],
    [23.513,-7.057],
    [23.823,-6.705],
    [24.01,-7.065],
    [24.273,-7.05],
]

def build(p: dict) -> cq.Workplane:
    thickness = float(p["thickness"])
    sx = float(p["length_scale"])
    sy = float(p["width_scale"])
    # topology parameter used to control optional smoothing (integer count)
    smooth_count = int(p["smooth_count"])

    # apply scale to base points (keeps centred at origin)
    pts = [(x * sx, y * sy) for x, y in BASE_POINTS]

    # optional simple smoothing by averaging with neighbours when smooth_count >1
    # keeps overall shape but uses topology param in geometry pipeline
    for _ in range(max(0, smooth_count - 1)):
        new_pts = []
        n = len(pts)
        for i in range(n):
            x0, y0 = pts[i]
            x1, y1 = pts[(i - 1) % n]
            x2, y2 = pts[(i + 1) % n]
            # light averaging, preserves extents
            nx = 0.5 * x0 + 0.25 * x1 + 0.25 * x2
            ny = 0.5 * y0 + 0.25 * y1 + 0.25 * y2
            new_pts.append((nx, ny))
        pts = new_pts

    wp = cq.Workplane("XY")
    wp = wp.moveTo(pts[0][0], pts[0][1])
    for x, y in pts[1:]:
        wp = wp.lineTo(x, y)
    wp = wp.close()
    # solid extrude from mounting face Z=0 upward +Z to thickness, centred in X/Y already
    result = wp.extrude(thickness, combine=True)
    return result

result = build({k: v[0] for k, v in PARAMS.items()})
