import cadquery as cq
import math

PARAMS = {
    "brick_length": (31.8, 30.0, 33.0, 0.1, "mm", True, "Brick length X"),
    "brick_width": (15.8, 14.5, 17.0, 0.1, "mm", True, "Brick width Y"),
    "thickness": (9.6, 8.0, 11.0, 0.1, "mm", True, "Body thickness Z"),
    "stud_dia": (4.8, 4.0, 5.2, 0.1, "mm", True, "Stud outer diameter"),
    "stud_height": (1.8, 1.2, 2.4, 0.1, "mm", True, "Stud height"),
    "stud_pitch_x": (8.0, 7.0, 9.0, 0.1, "mm", True, "Stud pitch X"),
    "stud_pitch_y": (8.0, 7.0, 9.0, 0.1, "mm", True, "Stud pitch Y"),
    "studs_x_count": (4, 1, 6, 1, "", False, "Studs along X"),
    "studs_y_count": (2, 1, 4, 1, "", False, "Studs along Y"),
}

def build(p: dict) -> cq.Workplane:
    brick_length = float(p["brick_length"])
    brick_width = float(p["brick_width"])
    thickness = float(p["thickness"])
    stud_dia = float(p["stud_dia"])
    stud_height = float(p["stud_height"])
    stud_pitch_x = float(p["stud_pitch_x"])
    stud_pitch_y = float(p["stud_pitch_y"])
    studs_x_count = int(round(float(p["studs_x_count"])))
    studs_y_count = int(round(float(p["studs_y_count"])))

    body = cq.Workplane("XY").rect(brick_length, brick_width).extrude(thickness)

    if studs_x_count > 0 and studs_y_count > 0:
        span_x = (studs_x_count - 1) * stud_pitch_x
        span_y = (studs_y_count - 1) * stud_pitch_y
        for ix in range(studs_x_count):
            x = -span_x / 2.0 + ix * stud_pitch_x
            for iy in range(studs_y_count):
                y = -span_y / 2.0 + iy * stud_pitch_y
                stud = (
                    cq.Workplane("XY")
                    .workplane(offset=thickness)
                    .center(x, y)
                    .circle(stud_dia / 2.0)
                    .extrude(stud_height)
                )
                body = body.union(stud)

    return body

result = build({k: v[0] for k, v in PARAMS.items()})
