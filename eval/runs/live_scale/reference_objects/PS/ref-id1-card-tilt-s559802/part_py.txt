import cadquery as cq
import math

PARAMS = {
    "card_length": (85.65, 80.0, 90.0, 0.1, "mm", True, "Card Length (X)"),
    "card_width": (54.02, 50.0, 58.0, 0.1, "mm", True, "Card Width (Y)"),
    "thickness": (0.8, 0.4, 1.5, 0.05, "mm", True, "Card Thickness (Z)"),
    "corner_radius": (3.18, 1.0, 5.0, 0.1, "mm", False, "Corner Radius"),
    "seg_count": (16, 8, 32, 1, "", False, "Fillet Segments"),
}

def build(p: dict) -> cq.Workplane:
    L = float(p["card_length"])
    W = float(p["card_width"])
    T = float(p["thickness"])
    R = float(p["corner_radius"])
    R = min(R, L/2 - 0.01, W/2 - 0.01)
    R = max(R, 0.1)
    base = cq.Workplane("XY").box(L, W, T, centered=True).translate((0, 0, T/2))
    try:
        result = base.edges("|Z").fillet(R)
    except Exception:
        result = base
    return result

result = build({k: v[0] for k, v in PARAMS.items()})
