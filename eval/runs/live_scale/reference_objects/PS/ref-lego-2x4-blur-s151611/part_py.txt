import cadquery as cq
import math

PARAMS = {
    # overall brick footprint
    "brick_length": (32.0, 30.0, 34.0, 0.1, "mm", False, "Brick length X"),
    "brick_width": (16.0, 14.0, 18.0, 0.1, "mm", False, "Brick width Y"),
    "thickness": (9.6, 8.0, 11.0, 0.1, "mm", False, "Brick body height Z"),
    # studs
    "stud_dia": (4.8, 4.0, 5.5, 0.1, "mm", True, "Stud outer diameter"),
    "stud_height": (1.8, 1.2, 2.4, 0.1, "mm", False, "Stud height above top"),
    "stud_pitch": (8.0, 7.0, 9.0, 0.1, "mm", True, "Stud pitch grid"),
    # topology - 2x4
    "n_x": (4, 1, 6, 1, "", False, "Studs along X"),
    "n_y": (2, 1, 4, 1, "", False, "Studs along Y"),
}

def build(p: dict) -> cq.Workplane:
    L = float(p["brick_length"])
    W = float(p["brick_width"])
    H = float(p["thickness"])
    Sd = float(p["stud_dia"])
    Sh = float(p["stud_height"])
    pitch = float(p["stud_pitch"])
    n_x = int(p["n_x"])
    n_y = int(p["n_y"])

    # Use measured polyline scaled to parametric L,W to keep fidelity to PartSpec
    # measured points from PartSpec
    pts_measured = [[-15.798,7.684],[-15.979,6.191],[-16.009,-3.012],[-15.82,-7.839],[15.511,-7.964],[15.915,-7.748],[16.003,7.34],[15.71,7.872],[-15.493,7.917]]
    # estimate measured extents
    xs = [pt[0] for pt in pts_measured]
    ys = [pt[1] for pt in pts_measured]
    Lm = max(xs) - min(xs)
    Wm = max(ys) - min(ys)
    # measured center offset
    cxm = (max(xs) + min(xs)) / 2.0
    cym = (max(ys) + min(ys)) / 2.0
    # scale and center points to requested L,W at origin
    sx = L / Lm if Lm != 0 else 1.0
    sy = W / Wm if Wm != 0 else 1.0
    pts = []
    for x, y in pts_measured:
        nx = (x - cxm) * sx
        ny = (y - cym) * sy
        pts.append((nx, ny))

    # body solid from Z=0 to H
    body = (
        cq.Workplane("XY")
        .workplane(offset=0)
        .moveTo(pts[0][0], pts[0][1])
        .polyline(pts[1:])
        .close()
        .extrude(H, combine=True)
    )

    # studs on top face, centered grid with pitch
    # positions: (i - (n_x-1)/2)*pitch , (j - (n_y-1)/2)*pitch
    result = body
    for ix in range(n_x):
        x = (ix - (n_x - 1) / 2.0) * pitch
        for iy in range(n_y):
            y = (iy - (n_y - 1) / 2.0) * pitch
            stud = (
                cq.Workplane("XY")
                .workplane(offset=H)
                .moveTo(x, y)
                .circle(Sd / 2.0)
                .extrude(Sh)
            )
            result = result.union(stud)

    return result

result = build({k: v[0] for k, v in PARAMS.items()})
