import cadquery as cq
import math

PARAMS = {
    "length": (31.859, 30.0, 33.0, 0.01, "mm", True, "Overall length X (long axis)"),
    "width": (23.918, 22.0, 25.0, 0.01, "mm", True, "Overall width Y"),
    "thickness": (2.1, 1.2, 3.0, 0.05, "mm", True, "Body thickness Z"),
    "chamfer_x": (3.26, 2.0, 6.0, 0.01, "mm", True, "Top-edge chamfer inset from right corner (X)"),
    "chamfer_y": (3.914, 2.0, 6.0, 0.01, "mm", True, "Right-edge chamfer inset from top corner (Y)"),
    "segments": (1, 1, 4, 1, "", False, "Topology segments per edge"),
}

def build(p: dict) -> cq.Workplane:
    length = float(p["length"])
    width = float(p["width"])
    thickness = float(p["thickness"])
    chamfer_x = float(p["chamfer_x"])
    chamfer_y = float(p["chamfer_y"])
    segments = int(p["segments"])

    chamfer_x = min(chamfer_x, length * 0.45)
    chamfer_y = min(chamfer_y, width * 0.45)

    hl = length / 2.0
    hw = width / 2.0

    base_pts = [
        (hl, hw - chamfer_y),
        (hl - chamfer_x, hw),
        (-hl, hw),
        (-hl, -hw),
        (hl, -hw),
    ]

    if segments > 1:
        expanded = []
        n = len(base_pts)
        for i in range(n):
            a = base_pts[i]
            b = base_pts[(i + 1) % n]
            expanded.append(a)
            for s in range(1, segments):
                t = s / float(segments)
                expanded.append((a[0] + (b[0] - a[0]) * t, a[1] + (b[1] - a[1]) * t))
        base_pts = expanded

    solid = cq.Workplane("XY").polyline(base_pts).close().extrude(thickness, both=False)
    return solid

result = build({k: v[0] for k, v in PARAMS.items()})
