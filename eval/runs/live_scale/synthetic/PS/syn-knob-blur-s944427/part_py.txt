import cadquery as cq
import math

PARAMS = {
    "thickness": (15.0, 8.0, 20.0, 0.1, "mm", False, "Body thickness"),
    "bore_dia": (6.0, 4.0, 8.0, 0.1, "mm", True, "D-shaft diameter"),
    "flat_depth": (1.0, 0.5, 2.0, 0.1, "mm", True, "Flat depth from circle"),
    "clearance": (0.2, 0.0, 0.5, 0.05, "mm", True, "Diametral fit clearance"),
    "bore_depth": (10.0, 5.0, 14.5, 0.5, "mm", True, "Blind bore depth from Z=0"),
    "top_bevel": (0.6, 0.0, 1.5, 0.1, "mm", False, "Top edge chamfer"),
    "resolve_n": (32, 8, 64, 1, "", False, "Arc resolution steps"),
}

def build(p: dict) -> cq.Workplane:
    thickness = float(p["thickness"])
    bore_dia = float(p["bore_dia"])
    flat_depth = float(p["flat_depth"])
    clearance = float(p["clearance"])
    bore_depth = float(p["bore_depth"])
    top_bevel = float(p["top_bevel"])
    resolve_n = int(p["resolve_n"])

    max_bore = max(0.5, thickness - 0.8)
    bore_depth = min(bore_depth, max_bore)

    pts = BASE_POINTS
    if resolve_n > 40:
        refined = []
        for i in range(len(pts)):
            a = pts[i]
            b = pts[(i+1) % len(pts)]
            refined.append(a)
            if math.hypot(b[0]-a[0], b[1]-a[1]) > 2.0:
                refined.append(((a[0]+b[0])*0.5, (a[1]+b[1])*0.5))
        pts = refined

    body = cq.Workplane("XY").polyline(pts).close().extrude(thickness)

    r_eff = (bore_dia + clearance) * 0.5
    flat_depth = max(0.1, min(flat_depth, r_eff * 1.6))
    fd = r_eff - flat_depth
    fd = max(-r_eff + 0.2, min(fd, r_eff - 0.1))
    x0_sq = r_eff * r_eff - fd * fd
    x0 = math.sqrt(max(0.0, x0_sq))

    if x0 > 0.05 and bore_depth > 0.2:
        d_shape = (
            cq.Workplane("XY")
            .moveTo(x0, fd)
            .lineTo(-x0, fd)
            .threePointArc((0, -r_eff), (x0, fd))
            .close()
            .extrude(bore_depth)
        )
        body = body.cut(d_shape)

    if top_bevel > 0.01:
        try:
            body = body.faces(">Z").chamfer(top_bevel)
        except Exception:
            pass

    return body

result = build({k: v[0] for k, v in PARAMS.items()})
