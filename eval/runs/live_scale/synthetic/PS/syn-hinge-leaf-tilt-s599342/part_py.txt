# Hole centers from PartSpec
HOLE1_CENTER = [-23.907, 5.22]
HOLE2_CENTER = [22.848, 5.093]

PARAMS = {
    "thickness": (3.0, 1.0, 5.0, 0.1, "mm", False, "Leaf thickness"),
    "hole1_dia": (3.618, 2.5, 5.0, 0.01, "mm", True, "Hole 1 diameter"),
    "hole2_dia": (3.37, 2.5, 5.0, 0.01, "mm", True, "Hole 2 diameter"),
    "csk_dia": (6.8, 5.0, 9.0, 0.1, "mm", True, "Countersink head diameter"),
    "csk_angle_deg": (90.0, 60.0, 120.0, 1.0, "deg", True, "Countersink angle"),
    "mount_hole_count": (2, 1, 2, 1, "", False, "Number of mounting holes"),
}

def build(p: dict) -> cq.Workplane:
    # Workplane on mounting face Z=0, +Z up
    wp = cq.Workplane("XY")
    # Build closed polyline from measured points
    wp = wp.moveTo(BASE_POINTS[0][0], BASE_POINTS[0][1])
    for pt in BASE_POINTS[1:]:
        wp = wp.lineTo(pt[0], pt[1])
    wp = wp.close()
    # Extrude solid to thickness - body is solid over full footprint
    solid = wp.extrude(p["thickness"], combine=True)

    # Decide which holes to create based on topology parameter
    n = int(p["mount_hole_count"])
    centers = []
    dias = []
    if n >= 1:
        centers.append(HOLE1_CENTER)
        dias.append(p["hole1_dia"])
    if n >= 2:
        centers.append(HOLE2_CENTER)
        dias.append(p["hole2_dia"])

    # Cut through holes with countersink on top face (+Z)
    # Using cskHole for countersunk mounting screws
    result = solid
    for ctr, dia in zip(centers, dias):
        result = (
            result.faces(">Z")
            .workplane()
            .pushPoints([tuple(ctr)])
            .cskHole(dia, p["csk_dia"], p["csk_angle_deg"])
        )

    return result

result = build({k: v[0] for k, v in PARAMS.items()})
