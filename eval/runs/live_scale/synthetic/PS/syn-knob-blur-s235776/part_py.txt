import cadquery as cq
import math

PARAMS = {
    "thickness": (13.5, 8.0, 20.0, 0.1, "mm", False, "Body thickness"),
    "outer_scale": (1.0, 0.8, 1.3, 0.01, "", False, "Outer profile scale"),
    "bore_dia": (6.0, 4.0, 8.0, 0.1, "mm", True, "Shaft diameter"),
    "flat_depth": (1.0, 0.5, 2.0, 0.1, "mm", True, "D flat depth from OD"),
    "clearance": (0.2, 0.0, 0.6, 0.05, "mm", True, "Fit clearance"),
    "bore_depth": (10.0, 5.0, 13.5, 0.5, "mm", False, "Blind bore depth from bottom"),
    "arc_facets": (64, 16, 128, 1, "", False, "Bore circle facets"),
}

# measured polyline footprint, roughly centered at origin
BASE_POINTS = [
    (12.996,4.967),(12.437,6.122),(11.358,6.698),(10.937,8.465),(10.38,9.271),(9.594,9.901),(8.095,10.109),(7.464,11.571),(6.556,12.234),(5.811,12.478),(4.864,12.153),(4.275,12.211),(3.169,13.395),(2.519,13.652),(1.474,13.707),(0.107,12.94),(-0.547,13.075),(-1.436,13.725),(-2.775,13.573),(-4.212,12.24),(-5.823,12.506),(-6.679,12.171),(-7.378,11.647),(-7.976,10.207),(-9.828,9.805),(-10.853,8.577),(-11.314,6.754),(-12.55,5.967),(-12.976,5.063),(-12.924,4.359),(-13.341,3.716),(-13.36,2.471),(-13.763,1.845),(-13.79,0.479),(-13.303,-0.449),(-13.609,-2.285),(-13.33,-3.658),(-13.051,-4.168),(-12.028,-4.972),(-12.019,-6.687),(-11.258,-7.95),(-9.58,-8.747),(-9.157,-10.172),(-7.766,-11.275),(-6.127,-11.437),(-5.05,-12.774),(-3.728,-13.171),(-2.109,-12.806),(-0.924,-13.621),(0.797,-13.66),(2.063,-12.811),(3.617,-13.194),(5.205,-12.638),(6.107,-11.452),(7.768,-11.289),(8.757,-10.618),(9.146,-10.119),(9.476,-8.85),(11.055,-8.14),(11.891,-6.959),(11.942,-5.176),(13.157,-4.066),(13.585,-2.482),(13.296,-0.496),(13.785,0.386),(13.821,1.117),(13.419,2.4),(13.468,3.153),(12.951,4.297),
]

def build(p: dict) -> cq.Workplane:
    thickness = float(p["thickness"])
    scale = float(p["outer_scale"])
    bore_dia = float(p["bore_dia"])
    flat_depth = float(p["flat_depth"])
    clearance = float(p["clearance"])
    bore_depth = float(p["bore_depth"])
    # clamp bore depth to not exceed thickness
    if bore_depth > thickness - 0.2:
        bore_depth = thickness - 0.2
    if bore_depth < 0.5:
        bore_depth = 0.5

    pts = [(x * scale, y * scale) for x, y in BASE_POINTS]

    # solid body extruded from mounting face Z=0 upward +Z
    wp = cq.Workplane("XY")
    wp = wp.moveTo(pts[0][0], pts[0][1])
    for x, y in pts[1:]:
        wp = wp.lineTo(x, y)
    wp = wp.close()
    body = wp.extrude(thickness, both=False)

    # D-shaft bore on underside, centered at origin, long axis +X
    # effective radius includes clearance
    r_eff = (bore_dia + clearance) * 0.5
    # flat offset from center along +Y (flat normal = +Y)
    flat_y = r_eff - flat_depth
    # clamp flat_y inside circle
    if flat_y > r_eff - 0.05:
        flat_y = r_eff - 0.05
    if flat_y < -r_eff + 0.05:
        flat_y = -r_eff + 0.05

    # chord half width
    v = r_eff * r_eff - flat_y * flat_y
    if v < 0:
        v = 0
    w = math.sqrt(v)

    # D profile: flat at y=flat_y, arc around bottom (major segment)
    # traverse from right flat intersection to left then arc back via bottom
    d_wp = cq.Workplane("XY")
    d_wp = d_wp.moveTo(w, flat_y)
    d_wp = d_wp.lineTo(-w, flat_y)
    # threePointArc via bottom point (0, -r_eff) back to start
    d_wp = d_wp.threePointArc((0, -r_eff), (w, flat_y))
    d_wp = d_wp.close()
    bore_tool = d_wp.extrude(bore_depth, both=False)

    # cut blind bore from bottom face (Z=0) upward
    body = body.cut(bore_tool)

    return body

result = build({k: v[0] for k, v in PARAMS.items()})
