import math

PARAMS = {
    "thickness": (14.5, 10.0, 20.0, 0.1, "mm", False, "Body Thickness"),
    "bore_dia": (6.0, 4.0, 8.0, 0.1, "mm", True, "D-Shaft Diameter"),
    "flat_depth": (0.8, 0.3, 1.5, 0.1, "mm", True, "Flat Depth"),
    "clearance": (0.2, 0.05, 0.4, 0.05, "mm", True, "Print Clearance"),
    "bore_depth": (12.0, 8.0, 14.0, 0.5, "mm", False, "Bore Depth"),
    "chamfer": (0.6, 0.0, 1.5, 0.1, "mm", False, "Top Edge Chamfer"),

}

PROFILE = [
    (-13.821,12.525),(-15.212,10.791),(-15.252,9.764),(-15.928,9.417),(-16.006,8.196),(-16.96,7.694),(-17.941,4.974),(-17.58,3.682),(-18.562,1.569),(-18.515,-1.563),(-17.55,-3.069),(-17.94,-4.847),(-16.824,-7.809),(-15.428,-8.976),(-15.116,-10.81),(-13.305,-12.977),(-11.519,-13.684),(-10.544,-15.276),(-7.854,-16.797),(-6.123,-16.827),(-4.594,-17.996),(-1.793,-18.48),(0.228,-17.947),(1.929,-18.481),(4.564,-18.007),(6.249,-16.808),(7.857,-16.816),(10.518,-15.305),(11.017,-14.241),(11.546,-13.696),(13.07,-13.145),(13.764,-12.539),(15.108,-10.813),(15.376,-9.104),(16.803,-7.864),(17.292,-6.907),(17.932,-4.934),(17.521,-3.284),(18.541,-1.506),(18.634,0.285),(18.506,1.781),(17.577,3.521),(17.952,4.994),(17.022,7.678),(16.247,8.093),(16.24,9.009),(15.26,9.667),(15.117,10.996),(13.627,12.767),(11.738,13.335),(10.729,15.23),(8.256,16.683),(6.186,16.582),(5.468,17.031),(4.774,17.933),(3.524,18.275),(1.842,18.468),(0.733,17.808),(-0.012,17.67),(-1.921,18.481),(-4.595,17.991),(-5.428,16.985),(-6.19,16.56),(-7.931,16.754),(-10.667,15.206),(-11.501,13.459),(-12.219,13.116),(-13.246,13.012)

def build(p: dict) -> cq.Workplane:
    thickness = float(p["thickness"])
    bore_dia = float(p["bore_dia"])
    flat_depth = float(p["flat_depth"])
    clearance = float(p["clearance"])
    bore_depth = float(p["bore_depth"])
    chamf = float(p["chamfer"])
    if bore_depth > thickness - 0.2:
        bore_depth = thickness - 0.2
    if bore_depth < 0.5:
        bore_depth = 0.5

    body = cq.Workplane("XY").polyline(PROFILE).close().extrude(thickness, combine=True)

    R = bore_dia / 2.0 + clearance / 2.0
    y0 = R - flat_depth
    if y0 > R:
        y0 = R - 0.01
    if y0 < -R:
        y0 = -R + 0.01

    kn = int(p.get("knurl_count", 32))
    arc_steps = max(24, kn)

    val = R * R - y0 * y0
    if val < 0:
        val = 0
    x0 = math.sqrt(val)

    pts = []
    pts.append((x0, y0))
    a_start = math.atan2(y0, x0)
    a_end = math.atan2(y0, -x0)
    a_end2 = a_end - 2.0 * math.pi
    for i in range(1, arc_steps):
        t = i / float(arc_steps)
        a = a_start + (a_end2 - a_start) * t
        x = R * math.cos(a)
        y = R * math.sin(a)
        pts.append((x, y))
    pts.append((-x0, y0))

    cutter = cq.Workplane("XY").polyline(pts).close().extrude(bore_depth, combine=True)

    body = body.cut(cutter)

    if chamf > 0.01:
        try:
            body = body.faces(">Z").wires().chamfer(chamf)
        except Exception:
            try:
                body = body.edges(cq.selectors.BoxSelector(( -20,-20,thickness-0.5),(20,20,thickness+0.5))).chamfer(chamf)
            except Exception:
                pass

    return body

result = build({k: v[0] for k, v in PARAMS.items()})
