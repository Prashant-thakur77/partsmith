# Hole centers from PartSpec
HOLE_CENTERS = {
    "hole_1": (-15.645, 8.567),
    "hole_2": (-8.413, -15.496),
    "hole_3": (0.0, 0.151),
    "hole_4": (8.409, 15.795),
    "hole_5": (15.639, -8.266),
}

def build(p: dict) -> cq.Workplane:
    thickness = float(p["thickness"])
    clearance = float(p["clearance"])
    center_dia = float(p["center_bore_dia"]) + float(clearance)
    # outer diameters include clearance for fit
    h1_dia = float(p["hole_1_dia"]) + float(clearance)
    h2_dia = float(p["hole_2_dia"]) + float(clearance)
    h4_dia = float(p["hole_4_dia"]) + float(clearance)
    h5_dia = float(p["hole_5_dia"]) + float(clearance)
    outer_count = int(p["outer_hole_count"])

    # Solid wheel body extruded from Z=0 to thickness, centred in X/Y by profile
    base = (
        cq.Workplane("XY")
        .polyline(PROFILE_POINTS)
        .close()
        .extrude(thickness, combine=True)
    )

    # Center bore - through hole
    cx, cy = HOLE_CENTERS["hole_3"]
    base = (
        base.faces(">Z").workplane()
        .pushPoints([(cx, cy)])
        .circle(center_dia / 2.0)
        .cutThruAll()
    )

    # Outer holes list in order
    outer_holes = [
        (HOLE_CENTERS["hole_1"][0], HOLE_CENTERS["hole_1"][1], h1_dia),
        (HOLE_CENTERS["hole_2"][0], HOLE_CENTERS["hole_2"][1], h2_dia),
        (HOLE_CENTERS["hole_4"][0], HOLE_CENTERS["hole_4"][1], h4_dia),
        (HOLE_CENTERS["hole_5"][0], HOLE_CENTERS["hole_5"][1], h5_dia),
    ]

    # Apply topology parameter: only first N outer holes
    for i in range(min(outer_count, len(outer_holes))):
        x, y, d = outer_holes[i]
        base = (
            base.faces(">Z").workplane()
            .pushPoints([(x, y)])
            .circle(d / 2.0)
            .cutThruAll()
        )

    return base

result = build({k: v[0] for k, v in PARAMS.items()})
